Пример .env
¶
# AgenticOS .env
# --- LLM providers ---
OPENROUTER_API_KEY=sk-or-v1-...
ANTHROPIC_API_KEY=sk-ant-...      # опционально, direct Anthropic
GEMINI_API_KEY=AIza...              # для multimodal и embeddings
# --- Observability ---
LANGFUSE_PUBLIC_KEY=pk-lf-...
LANGFUSE_SECRET_KEY=sk-lf-...
LANGFUSE_HOST=http://localhost:3000
# --- Persistence ---
AGENTICOS_POSTGRES_DSN=postgresql://agenticos:agenticos@localhost:5432/agenticos
SQLITE_PATH=agenticos.db
# --- Telegram driver ---
AGENTICOS_TG_BOT_TOKEN=1234567890:AAFmK...
AGENTICOS_TG_WORKSPACE_ROOT=/var/lib/agenticos
AGENTICOS_TG_MULTI_TENANT=1
AGENTICOS_TG_TRANSPORT=webhook
# --- Sidecars ---
AGENTICOS_GOTENBERG_URL=http://gotenberg:3000
AGENTICOS_DOCGEN_URL=http://docgen:3001
# --- Tool APIs ---
PERPLEXITY_API_KEY=pplx-...
ELEVENLABS_API_KEY=xi-...
